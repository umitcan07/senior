# Extra deps for the offline dev/verify harness (mod/dev/), NOT for the production worker.
# Install once into the running dev worker (the image doesn't ship these):
#   docker compose -f docker-compose.dev.yml exec worker-assessment \
#     pip install -r /worker/dev/requirements-dev.txt
#
# openai-whisper is the ASR backend verify.py uses for clean English transcripts (automap +
# asr). It runs on the container's existing PyTorch/CUDA 11.8 (faster-whisper's ctranslate2
# wants CUDA 12, so we use openai-whisper instead). Set VERIFY_ASR=powsm to fall back to
# POWSM's built-in ASR and skip this dependency.
openai-whisper

# panphon: articulatory feature table. As of E7.6 (#57) it is a PRODUCTION dependency
# (mod/phone_features.py — declared in mod/assessment/requirements.txt) and is also used
# by the feasibility POC (mod/dev/poc_feature_diff.py). Listed here too so the offline
# dev/verify harness can compute feature costs on a host. In the Linux dev worker
# `pip install panphon` works as-is. On a Windows host the editdistance/pandas-CSV build
# can be blocked by Application Control; only FeatureTable is needed, so there you can
# `pip install --no-deps panphon numpy pandas pyyaml regex unicodecsv munkres setuptools`
# and run python with `-X utf8` (panphon opens its IPA CSV without an explicit encoding).
panphon
